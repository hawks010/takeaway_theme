// In TTOS_Banner::init() or similar:
add_action( 'wp_footer', array( $this, 'render_banner' ) );

public function render_banner() {
    if ( ! $this->is_banner_enabled() ) return;
    if ( ! $this->should_show_now() ) return;
    $content = tt_content( 'banner', 'text', '' );
    if ( ! $content ) return;
    $style = tt_content( 'banner', 'background_colour', 'var(--tt-primary)' );
    echo '<div class="ttos-banner" role="banner" style="background:' . esc_attr($style) . '">';
    echo '<div class="ttos-banner-inner">' . wp_kses_post( $content ) . '</div>';
    echo '<button class="ttos-banner-close" aria-label="Close banner" data-ttos-banner-close>&times;</button>';
    echo '</div>';
}
```

            var first = focusable[0], last = focusable[focusable.length - 1];
            if (e.shiftKey && document.activeElement === first) {
                e.preventDefault(); last.focus();
            } else if (!e.shiftKey && document.activeElement === last) {
                e.preventDefault(); first.focus();
            }
        }
    });
    // Auto focus close on open
    function openPopup() {
        popup.classList.add('ttos-popup--open');
        popup.setAttribute('aria-hidden', 'false');
        popupClose.focus();
    }
    function closePopup() {
        popup.classList.remove('ttos-popup--open');
        popup.setAttribute('aria-hidden', 'true');
        localStorage.setItem('ttos_popup_dismissed', '1');
    }
    popupClose.addEventListener('click', closePopup);
}
```
